import { readFile, writeFile } from "node:fs";
export function multiply(input:string,output:string ,multiplier: number){
 let data: string = await readFile(input, 'utf-8')
 let numbers: string[] = data.split('\n');
 let result: number[] = [];
 for(let i=0; i<numbers.length; ++i){
    let num: number = Number(numbers[i]);
    if(Number.isNaN(num)){
        throw new Error;
        
    }
    result.push(num
    multiplier);
 }
}
